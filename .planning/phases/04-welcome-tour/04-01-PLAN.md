    ```typescript
    export interface TourStep {
      stepIndex: number;          // 0-4 matching the 5-step sequence
      pageUrl: string;            // Route path e.g. "/clientes"
      targetSelector: string;     // CSS selector for the element, e.g. "[data-tour='step-1']"
      title: string;              // Tooltip heading (pt-BR)
      description: string;        // Tooltip body text (pt-BR)
    }

    export type TourStatus = 'idle' | 'active' | 'completed' | 'skipped';

    export interface TourState {
      status: TourStatus;
      currentStepIndex: number;   // 0-4 when active, -1 when idle/completed
      steps: TourStep[];
      isMobile: boolean;          // Computed from viewport width <= 640px
    }

    export interface TourActions {
      startTour: () => void;
      nextStep: () => void;
      prevStep: () => void;
      skipTour: () => void;
      goToStep: (index: number) => void;
      completeStep: (stepIndex: number) => void;
    }

    export type TourContextValue = TourState & TourActions;
    ```

  <action>
    Create/modify three files:

    **1. TourProvider.tsx** (`'use client'`) — The central tour state machine:
    - Props: `children: ReactNode`, optional `initialSteps?: TourStep[]` (empty by default, Plan 04-02 provides step configs)
    - **State machine** using `useReducer` (preferred for multiple state transitions):
      ```typescript
      type TourAction = 
        | { type: 'START' }
        | { type: 'NEXT' }
        | { type: 'PREV' }
        | { type: 'GO_TO'; index: number }
        | { type: 'COMPLETE_STEP'; stepIndex: number }
        | { type: 'SKIP' }
        | { type: 'SET_STEPS'; steps: TourStep[] }
        | { type: 'SET_MOBILE'; isMobile: boolean };
      ```
    - **Initial state:** `{ status: 'idle', currentStepIndex: 0, steps: [], isMobile: false }`
    - **Reducer transitions:**
      - `START` → status='active', currentStepIndex=0 (only if steps.length > 0 and isMobile)
      - `NEXT` → if last step → status='completed'; else currentStepIndex++
      - `PREV` → currentStepIndex-- (min 0)
      - `GO_TO` → set currentStepIndex (if steps[index] exists)
      - `COMPLETE_STEP` → marks step as "visited" (store in a Set<string>), then auto-advance if stepIndex === currentStepIndex
      - `SKIP` → status='skipped'
      - `SET_STEPS` → replace steps array
      - `SET_MOBILE` → update isMobile flag
    - **Mobile detection:** useEffect with `window.innerWidth <= 640`, debounced resize listener (300ms debounce)
    - **Rendering:** When `status === 'active' && isMobile`:
      - Render `TourOverlay` with current step's target element rect
      - Render `TourTooltip` positioned at the target element
      - Calculate `targetRect` by querying `document.querySelector(step.targetSelector)` and calling `getBoundingClientRect()`
      - Poll/recalculate position on scroll/resize via useEffect
    - **Context value:** Spread both state (`TourState`) and actions (`TourActions`) into context
    - Edge case: if target element is not found, still render tooltip centered at the top of the viewport with a note "Elemento não encontrado — navegue até a página correta"
