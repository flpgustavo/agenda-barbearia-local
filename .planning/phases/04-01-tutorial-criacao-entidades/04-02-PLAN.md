    ```
    - Props: totalSteps (number), currentStep (number)
    - Render: horizontal row of circles connected by lines
    - States per circle:
      - Active (i === currentStep): filled orange (#ff592c), scale-110, 12px circle
      - Completed (i < currentStep): filled gray (#62738d), Check icon (lucide-react)
      - Pending (i > currentStep): outlined circle, border #dce4eb
    - Progress lines between circles:
      - Filled portion: left of current step → orange
      - Empty portion: right of current step → dashed #dce4eb
    - Use motion from framer-motion for active dot scale animation
    - Tailwind classes from globals.css for colors (bg-primary, bg-muted, etc.)
    ```

    ```typescript
    'use client';

    import { motion } from 'framer-motion';
    import { Check } from 'lucide-react';
    import { cn } from '@/lib/utils';

    interface StepIndicatorProps {
      totalSteps: number;
      currentStep: number;
      className?: string;
    }

    export function StepIndicator({ totalSteps, currentStep, className }: StepIndicatorProps) {
      return (
        <div className={cn('flex items-center justify-center gap-0 w-full', className)}>
          {Array.from({ length: totalSteps }).map((_, i) => {
            const isActive = i === currentStep;
            const isCompleted = i < currentStep;
            const isPending = i > currentStep;
            const hasLineBefore = i > 0;
            const hasLineAfter = i < totalSteps - 1;

            return (
              <div key={i} className="flex items-center">
                {/* Line before */}
                {hasLineBefore && (
                  <div className={cn(
                    'w-8 h-[2px] mr-0',
                    i <= currentStep ? 'bg-primary' : 'border [border-top:dashed_2px] border-muted'
                  )} />
                )}

                {/* Circle */}
                {isActive ? (
                  <motion.div
                    initial={{ scale: 1 }}
                    animate={{ scale: 1.2 }}
                    transition={{ type: 'spring', stiffness: 300, damping: 20 }}
                    className="w-3 h-3 rounded-full bg-primary flex-shrink-0"
                  />
                ) : isCompleted ? (
                  <div className="w-3 h-3 rounded-full bg-muted-foreground flex-shrink-0 flex items-center justify-center">
                    <Check className="w-2 h-2 text-background" />
                  </div>
                ) : (
                  <div className="w-3 h-3 rounded-full border-2 border-muted flex-shrink-0" />
                )}

                {/* Line after */}
                {hasLineAfter && (
                  <div className={cn(
                    'w-8 h-[2px] ml-0',
                    i < currentStep ? 'bg-primary' : 'border [border-top:dashed_2px] border-muted'
                  )} />
                )}
              </div>
            );
          })}
        </div>
      );
    }
    ```

    Create src/components/tutorial/TutorialOverlay.tsx:

    ```
    - Props:
      - open: boolean
      - onOpenChange: (open: boolean) => void
      - drawerOpen: boolean (controls entity form drawer)
      - onDrawerOpenChange: (open: boolean) => void
      - onStepComplete: () => void (called when entity form is submitted)
    
    - Internal state: showFinishStep (boolean, true when all steps complete but dialog still open)
    
    - Step navigation flow (per RESEARCH.md):
      1. User clicks CTA → onDrawerOpenChange(true) → entity form drawer opens
      2. User submits form → onStepComplete() → completeStep(entity) → drawer closes → advance
      3. On last step completion → showFinishStep = true
    
    - When showFinishStep:
      - Title: "Tudo Pronto!" (UI-SPEC)
      - Description: "Você completou o tutorial. Comece a usar o sistema!" (UI-SPEC)
      - CTA: "Começar a Usar" → onOpenChange(false)
      - No back button on finish step
    ```

    ```typescript
    'use client';

    import { AnimatePresence, motion } from 'framer-motion';
    import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter } from '@/components/ui/dialog';
    import { Button } from '@/components/ui/button';
    import { StepIndicator } from './StepIndicator';
    import { useTutorial } from '@/hooks/useTutorial';
    import { TUTORIAL_STEPS, TutorialEntity } from '@/core/tutorial';

    interface TutorialOverlayProps {
      open: boolean;
      onOpenChange: (open: boolean) => void;
      drawerOpen: boolean;
      onDrawerOpenChange: (open: boolean) => void;
      onStepComplete: () => void;
    }

    export function TutorialOverlay({
      open,
      onOpenChange,
      drawerOpen,
      onDrawerOpenChange,
      onStepComplete,
    }: TutorialOverlayProps) {
      const { currentStep, stepIndex, totalSteps, completeStep, goToNextStep, goToPrevStep, dismissTutorial } = useTutorial();

      // Show finish step when at last index and no current step
      const showFinish = currentStep === null && stepIndex >= TUTORIAL_STEPS.length;

      // Animation variants for step content slide
      const slideVariants = {
        enter: (direction: number) => ({ x: direction > 0 ? 50 : -50, opacity: 0 }),
        center: { x: 0, opacity: 1 },
        exit: (direction: number) => ({ x: direction < 0 ? 50 : -50, opacity: 0 }),
      };

      // Handle CTA click: open drawer
      const handleCtaClick = () => {
        onDrawerOpenChange(true);
      };

      // Handle form success: mark step complete, close drawer
      const handleFormSuccess = () => {
        if (currentStep) {
          completeStep(currentStep.id as TutorialEntity);
        }
        onDrawerOpenChange(false);
        onStepComplete();
      };

      // When drawer closes after form submission, step was completed
      // Trigger tutorial to advance to next step or finish
      const handleDrawerChange = (drawerIsOpen: boolean) => {
        onDrawerOpenChange(drawerIsOpen);
      };

      // Skip tutorial
      const handleSkip = () => {
        dismissTutorial();
        onOpenChange(false);
      };

      // Finish tutorial
      const handleFinish = () => {
        onOpenChange(false);
      };

      // Get CTA from TUTORIAL_STEPS if current step exists
      const currentStepData = currentStep ?? TUTORIAL_STEPS[stepIndex];

      return (
        <Dialog open={open} onOpenChange={onOpenChange}>
          <DialogContent
            className="max-w-[340px] p-5 rounded-2xl shadow-neon"
            showCloseButton={true}
            onEscapeKeyDown={(e) => e.preventDefault()} // Allow ESC to close on tutorial
          >
            <AnimatePresence mode="wait" initial={false}>
              {showFinish ? (
                <motion.div
                  key="finish"
                  variants={slideVariants}
                  initial="enter"
                  animate="center"
                  exit="exit"
                  transition={{ duration: 0.3 }}
                >
                  <DialogHeader>
                    <DialogTitle className="text-xl font-semibold text-center">Tudo Pronto!</DialogTitle>
                    <DialogDescription className="text-center text-sm">
                      Você completou o tutorial. Comece a usar o sistema!
                    </DialogDescription>
                  </DialogHeader>
                  <div className="flex justify-center py-4">
                    <StepIndicator totalSteps={totalSteps} currentStep={totalSteps} />
                  </div>
                  <DialogFooter className="flex-col-reverse sm:flex-col sm:justify-center gap-2">
                    <Button onClick={handleFinish} className="w-full bg-primary hover:bg-primary/90">
                      Começar a Usar
                    </Button>
                  </DialogFooter>
                </motion.div>
              ) : currentStepData ? (
                <motion.div
                  key={currentStepData.id}
                  variants={slideVariants}
                  initial="enter"
                  animate="center"
                  exit="exit"
                  custom={1}
                  transition={{ duration: 0.3 }}
                >
                  <DialogHeader>
                    <div className="text-center text-xs text-muted-foreground uppercase tracking-wider">
                      Passo {stepIndex + 1} de {totalSteps}
                    </div>
                    <DialogTitle className="text-xl font-semibold text-center">
                      {currentStepData.title}
                    </DialogTitle>
                    <DialogDescription className="text-center text-sm">
                      {currentStepData.description}
                    </DialogDescription>
                  </DialogHeader>

                  <div className="flex justify-center py-4">
                    <StepIndicator totalSteps={totalSteps} currentStep={stepIndex} />
                  </div>

                  <div className="flex flex-col gap-3">
                    <Button
                      onClick={handleCtaClick}
                      className="w-full bg-primary hover:bg-primary/90"
                    >
                      {currentStepData.ctaLabel}
                    </Button>
                  </div>

                  <DialogFooter className="flex-col-reverse sm:flex-col sm:justify-center gap-2 pt-4">
                    <button
                      onClick={handleSkip}
                      className="text-xs text-muted-foreground hover:text-foreground underline-offset-4 hover:underline cursor-pointer bg-transparent border-none p-0"
                    >
                      Pular tutorial
                    </button>
                    <div className="flex gap-2">
                      {stepIndex > 0 && (
                        <Button
                          variant="outline"
                          onClick={goToPrevStep}
                          className="flex-1"
                        >
                          Voltar
                        </Button>
                      )}
                      {stepIndex < totalSteps - 1 && (
                        <Button
                          variant="outline"
                          onClick={goToNextStep}
                          className="flex-1"
                        >
                          Avançar
                        </Button>
                      )}
                    </div>
                  </DialogFooter>
                </motion.div>
              ) : null}
            </AnimatePresence>
          </DialogContent>
        </Dialog>
      );
    }

    // Also export a helper hook for page integration
    export function useTutorialOverlayConfig(entity: TutorialEntity) {
      const { currentStep, isStepComplete } = useTutorial();
      return {
        shouldShow: currentStep?.id === entity && !isStepComplete(entity),
      };
    }
    ```

    Note: The entity form drawer integration is handled in the page integration task (Plan 3).
    TutorialOverlay just triggers the drawer via `onDrawerOpenChange` and `onStepComplete`.
