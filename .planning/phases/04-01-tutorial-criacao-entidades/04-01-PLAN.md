    ```typescript
    'use client';

    import { useState, useEffect, useCallback } from 'react';
    import { TUTORIAL_STEPS, TUTORIAL_STORAGE_KEY, TutorialEntity, TutorialStep } from '@/core/tutorial';

    interface UseTutorialReturn {
      currentStep: TutorialStep | null;
      stepIndex: number;
      totalSteps: number;
      isStepComplete: (entity: TutorialEntity) => boolean;
      completeStep: (entity: TutorialEntity) => void;
      dismissTutorial: () => void;
      resetTutorial: () => void;
      goToNextStep: () => void;
      goToPrevStep: () => void;
      isComplete: boolean;
      isDismissed: boolean;
      isActive: boolean;
    }

    const initialState = {
      currentStepIndex: 0,
      completedKeys: [] as TutorialEntity[],
      dismissed: false,
    };

    export function useTutorial(): UseTutorialReturn {
      // Guard SSR: localStorage not available during server render
      const [state, setState] = useState(initialState);
      const [isHydrated, setIsHydrated] = useState(false);

      // Hydrate from localStorage on mount (client-side only)
      useEffect(() => {
        try {
          const stored = localStorage.getItem(TUTORIAL_STORAGE_KEY);
          if (stored) {
            setState(JSON.parse(stored));
          }
        } catch {
          // Ignore parse errors — use initial state
        }
        setIsHydrated(true);
      }, []);

      // Persist to localStorage whenever state changes (after hydration)
      useEffect(() => {
        if (isHydrated) {
          localStorage.setItem(TUTORIAL_STORAGE_KEY, JSON.stringify(state));
        }
      }, [state, isHydrated]);

      const currentStep = TUTORIAL_STEPS[state.currentStepIndex] ?? null;

      const isStepComplete = useCallback((entity: TutorialEntity): boolean => {
        return state.completedKeys.includes(entity);
      }, [state.completedKeys]);

      const completeStep = useCallback((entity: TutorialEntity) => {
        setState((prev) => {
          if (prev.completedKeys.includes(entity)) return prev;
          const nextIndex = prev.currentStepIndex + 1;
          const isLastStep = nextIndex >= TUTORIAL_STEPS.length;
          return {
            ...prev,
            completedKeys: [...prev.completedKeys, entity],
            currentStepIndex: isLastStep ? prev.currentStepIndex : nextIndex,
          };
        });
      }, []);

      const dismissTutorial = useCallback(() => {
        setState((prev) => ({
          ...prev,
          dismissed: true,
          // Mark all as complete so it never re-shows
          completedKeys: TUTORIAL_STEPS.map((s) => s.id),
        }));
      }, []);

      const resetTutorial = useCallback(() => {
        setState(initialState);
      }, []);

      const goToNextStep = useCallback(() => {
        setState((prev) => {
          const next = prev.currentStepIndex + 1;
          if (next >= TUTORIAL_STEPS.length) return prev;
          return { ...prev, currentStepIndex: next };
        });
      }, []);

      const goToPrevStep = useCallback(() => {
        setState((prev) => {
          if (prev.currentStepIndex === 0) return prev;
          return { ...prev, currentStepIndex: prev.currentStepIndex - 1 };
        });
      }, []);

      const isComplete = state.completedKeys.length >= TUTORIAL_STEPS.length;
      const isDismissed = state.dismissed || isComplete;

      return {
        currentStep,
        stepIndex: state.currentStepIndex,
        totalSteps: TUTORIAL_STEPS.length,
        isStepComplete,
        completeStep,
        dismissTutorial,
        resetTutorial,
        goToNextStep,
        goToPrevStep,
        isComplete,
        isDismissed,
        isActive: isHydrated && !isDismissed,
      };
    }
    ```
